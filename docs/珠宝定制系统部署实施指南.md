### 步骤3：创建GitHub Actions工作流
在项目根目录创建 `.github/workflows/deploy.yml`：

```yaml
name: Deploy Jewelry System

on:
  push:
    branches: [ main ]

env:
  REGISTRY: ghcr.io
  IMAGE_NAME: ${{ github.repository }}/jewelry-backend

jobs:
  build-and-push:
    runs-on: ubuntu-latest
    permissions:
      contents: read
      packages: write
    
    steps:
    - uses: actions/checkout@v3
    
    - name: Set up JDK 17
      uses: actions/setup-java@v3
      with:
        java-version: '17'
        distribution: 'temurin'
    
    - name: Build with Maven
      run: mvn -f backend/pom.xml clean package -DskipTests
    
    - name: Log in to GitHub Container Registry
      uses: docker/login-action@v2
      with:
        registry: ${{ env.REGISTRY }}
        username: ${{ github.actor }}
        password: ${{ secrets.GITHUB_TOKEN }}
    
    - name: Build and push Docker image
      uses: docker/build-push-action@v4
      with:
        context: ./backend
        push: true
        tags: ${{ env.REGISTRY }}/${{ env.IMAGE_NAME }}:latest
  
  deploy:
    needs: build-and-push
    runs-on: ubuntu-latest
    if: github.ref == 'refs/heads/main'
    
    steps:
    - uses: actions/checkout@v3
    
    - name: Deploy to Server
      uses: appleboy/ssh-action@v0.1.5
      with:
        host: ${{ secrets.SSH_HOST }}
        username: ${{ secrets.SSH_USERNAME }}
        password: ${{ secrets.SSH_PASSWORD }}
        script: |
          # 进入项目目录
          cd /mnt/newdisk/app/jewelry-system
          
          # 更新代码
          if [ ! -d ".git" ]; then
            git clone https://github.com/${{ github.repository }}.git .
          else
            git pull origin main
          fi
          
          # 创建docker-compose.prod.yml
          cat > docker-compose.prod.yml << 'EOF'
version: "3.9"

services:
  mysql:
    image: mysql:8.4
    container_name: jewelry-mysql
    restart: unless-stopped
    environment:
      MYSQL_ROOT_PASSWORD: @Group666
      MYSQL_DATABASE: moje_database
      TZ: Asia/Shanghai
    volumes:
      - ./data/mysql:/var/lib/mysql
      - ./backend/src/main/resources/db:/docker-entrypoint-initdb.d:ro
    command: ["--character-set-server=utf8mb4","--collation-server=utf8mb4_unicode_ci"]

  backend:
    image: ${{ env.REGISTRY }}/${{ env.IMAGE_NAME }}:latest
    container_name: jewelry-backend
    depends_on:
      mysql:
        condition: service_healthy
    environment:
      DB_HOST: mysql
      DB_PORT: 3306
      DB_NAME: moje_database
      DB_USER: root
      DB_PASSWORD: @Group666
      OSS_ACCESS_KEY_ID: ${{ secrets.OSS_ACCESS_KEY_ID }}
      OSS_ACCESS_KEY_SECRET: ${{ secrets.OSS_ACCESS_KEY_SECRET }}
      OSS_ENDPOINT: oss-cn-guangzhou.aliyuncs.com
      OSS_BUCKET: moje-jewelry-files
      SERVER_PORT: 8851
    ports:
      - "8851:8851"
    volumes:
      - ./backend/src/main/resources/db:/app/db:ro
      - ./logs:/app/logs
    restart: unless-stopped

EOF
          
          # 拉取镜像并启动服务
          docker compose -f docker-compose.prod.yml pull
          docker compose -f docker-compose.prod.yml up -d --force-recreate
          
          # 验证部署
          sleep 10
          echo "部署完成！服务地址: http://39.102.213.51:8851"
```
